Typst

Some miscellanous Typst utils I use:

// Self-numbering problems, subproblems and lemmas.
//
// prob() opens a numbered problem box and quietly files away its title and
// color. subprob() reopens that same box later with no arguments at all and
// appends ", contd.". lemma() numbers itself problem.lemma off the two
// counters, so reordering the problems renumbers every lemma on the next
// compile. thm() rides along for plain titled results. Needs showybox.
//
//   #prob("Stable matching")[
//     (a) Prove the algorithm terminates.
//     #lemma[Engaged receivers only ever trade up.][ ...proof... ]
//   ]
//   #subprob[ (b) Show the returned matching is stable. #qed ]

#import "@preview/showybox:2.0.2": showybox

#let problem_counter = counter("problem")       // problems: 1, 2, 3, ...
#let lemma_counter   = counter("lemma")         // lemmas within a problem
#let last_prob       = state("last_prob", (:))  // title + color, carried

// `title` is positional: prob("My title")[...], or prob(none)[...] to drop
// the title bar. The title and color are stashed in `last_prob` below.
#let prob(color: blue, title, count: true, ..body) = {
  if count [== Problem #problem_counter.step() #context problem_counter.display()]
  if count {
    last_prob.update(_ => (title: title, color: color))
    lemma_counter.update(1)            // restart lemma numbering
  }
  if title != none {
    showybox(
      frame: (
        border-color: black,
        title-color:  color.lighten(80%),
        body-color:   color.lighten(90%),
      ),
      title-style: (color: black, weight: "bold"),
      title: [#title],
      ..body,
    )
  } else {
    showybox(
      frame: (
        border-color: black,
        title-color:  color.lighten(80%),
        body-color:   color.lighten(90%),
      ),
      title-style: (color: black, weight: "bold"),
      ..body,
    )
  }
}

// No title, no color: subprob reads both back out of `last_prob`.
#let subprob(..body) = context {
  let color = last_prob.get().at("color", default: blue)
  showybox(
    frame: (
      border-color: black,
      title-color:  color.lighten(80%),
      body-color:   color.lighten(90%),
    ),
    title-style: (color: black, weight: "bold"),
    title: [#last_prob.get().at("title", default: ""), contd.],
    ..body,
  )
}

// Numbered problem.lemma straight off the two counters.
#let lemma(claim, color: green, ..body) = context {
  lemma_counter.step()
  showybox(
    frame: (body-color: color.lighten(90%)),
    [*#underline[Lemma #problem_counter.get().at(0).#lemma_counter.get().at(0):]* #claim],
    ..body,
  )
}

// A plain titled claim box, no numbering, for theorems and named results.
#let thm(name: "Theorem", statement, color: red, ..body) = showybox(
  frame: (body-color: color.lighten(90%)),
  [*#underline[#name:]* #statement],
  ..body,
)
// End-of-proof squares. Drop #qed at the end of a proof and a filled square
// snaps to the right margin. h(1fr) eats the space to its left; the box pins
// the height to 0.5em and shifts the baseline, so scaling the glyph up to
// 1.6em never stretches the line it lands on. #lemqed is the hollow version,
// for closing a lemma.
#let qed = h(1fr) + box(
  height: 0.5em, clip: false, baseline: -0.17em,
  align(horizon, text(1.6em, sym.square.filled)),
)
#let lemqed = h(1fr) + box(
  height: 0.5em, clip: false, baseline: -0.17em,
  align(horizon, text(1.6em, sym.square)),
)
// A slanted quotient, for when a flat fraction would read as division
// rather than as a relation. The numerator drifts up and to the right, the
// denominator down and to the left, and lr() grows the delimiters to wrap
// the slanted pair. `size` is bumped by the same offset so nothing clips.
#let relmod(s, r) = {
  set text(size: 11pt, stroke: 0.05pt)
  let offset = 3pt
  let num   = move(dy: -offset, dx:  offset, s)
  let denom = move(dy:  offset, dx: -offset, r)
  $display(lr(num mid(slash) denom, size: #(80% + 2 * offset)))$
}
// An algorithm block that measures its own header. A thin wrapper over the
// algo package; the idea worth keeping is the rule under the heading. It is
// drawn with measure(), so it comes out exactly as wide as the procedure
// name plus its parameter list, whatever you call it.
//
//   #alg(name: "Gcd", params: $a, b$)[
//     while $b != 0$:#i\
//     $(a, b) = (b, a mod b)$ #comment[Euclid's step]#d\
//     return $a$
//   ]
//
// #i and #d open and close an indent; a trailing \ ends a line.
#import "@preview/algo:0.3.6": algo, i, d, comment

#let alg(name: "", params: [], body) = [
  #set text(bottom-edge: -0.2em)
  #algo(
    header: [
      #set text(size: 14pt)
      #smallcaps(name)$($#params$)$:
      #v(-1.1em)
      #context(std.line(
        length: measure(name).width + measure(params).width + 2em,
      ))
    ],
    comment-prefix: [#sym.chevron.l.double~],
    comment-styles: (fill: rgb(0%, 50%, 0%)),
    indent-size: 15pt,
    indent-guides: 0.5pt + gray,
    row-gutter: 6pt,
    column-gutter: 5pt,
    inset: 5pt,
    stroke: 0.7pt + black,
    fill: gray.lighten(94%),
    strong-keywords: false,
  )[#body]
]
// Boxed answers. ans() frames a final answer and centers it on its own
// line. inlineans() does the same mid-sentence: outset lets the rule
// breathe past the text, baseline drops the box level with the words.
#let ans(body) = align(center)[#box(stroke: 0.7pt, inset: 0.7em)[#body]]
#let inlineans(body) = h(0.1em) + box(
  body,
  stroke: 0.7pt,
  inset: 0.25em,
  outset: (x: 0.1em, y: 0.5em),
  baseline: 0.30em,
)
// Small conveniences. inline() flushes a green chevroned note to the right
// margin; aside() greys a passage out; the last three are one-liners not
// worth retyping in every file.
#let inline(body) = [
  #set text(green.darken(30%))
  #h(1fr) $chevron.l.double$#body$chevron.r.double$]

#let aside(body) = [
  #set text(fill: gray)
  #body
]

#let proof = underline[*Proof:*]                     // a "Proof:" lead-in
#let hline = std.line(length: 100%, stroke: 0.5pt)   // a full-width rule
#let inner(a, b) = $lr(chevron.l #a, #b chevron.r)$  // inner product ⟨a, b⟩
