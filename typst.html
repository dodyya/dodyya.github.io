--- title: Typst ---

Typst

I write my problem sets and lecture notes in Typst, and over a few semesters these helpers piled up. The one I'm proudest of numbers problems, subproblems and lemmas by itself, and keeps numbering them correctly when I shuffle things around. Everything below is free to take, whole or piece by piece.

Self-numbering problems, subproblems and lemmas

prob opens a numbered problem box and quietly records its title and color. subprob reopens that same box later with no arguments at all, appending contd., and lemma numbers itself problem.lemma off the two counters, so reordering problems renumbers every lemma on the next compile. thm rides along for plain titled results. It all needs the showybox package.

#import "@preview/showybox:2.0.2": showybox

// Two counters and one state value carry everything between calls.
#let problem_counter = counter("problem")  // counts problems: 1, 2, 3, ...
#let lemma_counter   = counter("lemma")    // counts lemmas within a problem
#let last_prob       = state("last_prob", (:))  // title + color of the open one

// A numbered problem box. `title` is positional, so call it as
//   #prob("My title")[ ... ]   -- or   #prob(none)[ ... ]   for no title bar.
// The title and color are stashed in `last_prob` for the helpers below.
#let prob(color: blue, title, count: true, ..body) = {
  if count [== Problem #problem_counter.step() #context problem_counter.display()]
  if count {
    last_prob.update(_ => (title: title, color: color))
    lemma_counter.update(1)            // restart lemma numbering each problem
  }
  if title != none {
    showybox(
      frame: (
        border-color: black,
        title-color:  color.lighten(80%),
        body-color:   color.lighten(90%),
      ),
      title-style: (color: black, weight: "bold"),
      title: [#title],
      ..body,
    )
  } else {
    showybox(
      frame: (
        border-color: black,
        title-color:  color.lighten(80%),
        body-color:   color.lighten(90%),
      ),
      title-style: (color: black, weight: "bold"),
      ..body,
    )
  }
}

// Continues the problem above. No title, no color: it reads both back out
// of `last_prob` and appends ", contd." on its own.
#let subprob(..body) = context {
  let color = last_prob.get().at("color", default: blue)
  showybox(
    frame: (
      border-color: black,
      title-color:  color.lighten(80%),
      body-color:   color.lighten(90%),
    ),
    title-style: (color: black, weight: "bold"),
    title: [#last_prob.get().at("title", default: ""), contd.],
    ..body,
  )
}

// A lemma, numbered as problem.lemma -- the second lemma inside problem 3
// prints "Lemma 3.2". Reorder the problems and it all renumbers.
#let lemma(claim, color: green, ..body) = context {
  lemma_counter.step()
  showybox(
    frame: (body-color: color.lighten(90%)),
    [*#underline[Lemma #problem_counter.get().at(0).#lemma_counter.get().at(0):]* #claim],
    ..body,
  )
}

// A plain titled claim box, no numbering -- for theorems and named results.
#let thm(name: "Theorem", statement, color: red, ..body) = showybox(
  frame: (body-color: color.lighten(90%)),
  [*#underline[#name:]* #statement],
  ..body,
)

In a document it reads about as plainly as the problem set itself:

#prob("Stable matching")[
  (a) Prove that the Gale-Shapley algorithm always terminates.

  #lemma[Once engaged, a receiver only ever trades up.][
    A receiver swaps partners solely for a more-preferred proposer. #lemqed
  ]
]

#subprob[
  (b) Show the matching it returns is stable. #qed
]

End-of-proof squares

Drop #qed at the end of a proof and a filled square snaps to the right margin without disturbing the line it lands on. #lemqed is the hollow square, for closing a lemma.

// `h(1fr)` swallows the space to its left, flushing the square to the right
// margin. The box pins its height to 0.5em and tweaks the baseline, so
// scaling the glyph up to 1.6em never stretches the line it sits on.
#let qed = h(1fr) + box(
  height: 0.5em, clip: false, baseline: -0.17em,
  align(horizon, text(1.6em, sym.square.filled)),
)

// The hollow version, for the end of a lemma's sub-proof.
#let lemqed = h(1fr) + box(
  height: 0.5em, clip: false, baseline: -0.17em,
  align(horizon, text(1.6em, sym.square)),
)

A slanted quotient

A quotient-style notation where the two halves are nudged apart along a diagonal and the brackets grow to wrap them. I reach for it when a flat fraction would read as division rather than as a relation.

// The numerator drifts up and to the right, the denominator down and to the
// left, and `lr` grows the delimiters to wrap the slanted pair. `size` is
// bumped by the same offset so the brackets never clip the moved content.
#let relmod(s, r) = {
  set text(size: 11pt, stroke: 0.05pt)
  let offset = 3pt
  let num   = move(dy: -offset, dx:  offset, s)
  let denom = move(dy:  offset, dx: -offset, r)
  $display(lr(num mid(slash) denom, size: #(80% + 2 * offset)))$
}

An algorithm block that measures its own header

A thin wrapper over the algo package. The trick worth stealing is the rule under the heading: it is drawn with measure(), so it always comes out exactly as wide as the procedure name and its argument list.

#import "@preview/algo:0.3.6": algo, i, d, comment

// The one nice idea here is the underline in the header: `measure()` reports
// the width of the name and the params, and `std.line` is drawn to match.
#let alg(name: "", params: [], body) = [
  #set text(bottom-edge: -0.2em)
  #algo(
    header: [
      #set text(size: 14pt)
      #smallcaps(name)$($#params$)$:
      #v(-1.1em)
      #context(std.line(
        length: measure(name).width + measure(params).width + 2em,
      ))
    ],
    comment-prefix: [#sym.chevron.l.double~],
    comment-styles: (fill: rgb(0%, 50%, 0%)),
    indent-size: 15pt,
    indent-guides: 0.5pt + gray,
    row-gutter: 6pt,
    column-gutter: 5pt,
    inset: 5pt,
    stroke: 0.7pt + black,
    fill: gray.lighten(94%),
    strong-keywords: false,
  )[#body]
]

Using it:

// #i and #d open and close an indent; a trailing `\` ends a line.
#alg(name: "Gcd", params: $a, b$)[
  while $b != 0$:#i\
  $(a, b) = (b, a mod b)$ #comment[Euclid's step]#d\
  return $a$
]

Boxed answers

ans frames a final answer and centers it on its own line. inlineans does the same mid-sentence, using outset so the rule clears the text and baseline so the box sits level with the words around it.

// A boxed final answer, centered on its own line.
#let ans(body) = align(center)[#box(stroke: 0.7pt, inset: 0.7em)[#body]]

// The same idea inline. `outset` lets the rule breathe past the text;
// `baseline` drops the box so it sits level with the surrounding words.
#let inlineans(body) = h(0.1em) + box(
  body,
  stroke: 0.7pt,
  inset: 0.25em,
  outset: (x: 0.1em, y: 0.5em),
  baseline: 0.30em,
)

Small conveniences

The leftovers. inline flushes a green chevroned note to the right margin, aside greys a passage out, and the last three are one-liners not worth retyping in every file.

// A green note in heavy chevrons, flushed to the right margin.
#let inline(body) = [
  #set text(green.darken(30%))
  #h(1fr) $chevron.l.double$#body$chevron.r.double$]

// Greys a passage out -- for remarks meant to read as secondary.
#let aside(body) = [
  #set text(fill: gray)
  #body
]

// Three one-liners worth never retyping:
#let proof = underline[*Proof:*]                     // a "Proof:" lead-in
#let hline = std.line(length: 100%, stroke: 0.5pt)   // a full-width rule
#let inner(a, b) = $lr(chevron.l #a, #b chevron.r)$  // inner product ⟨a, b⟩
